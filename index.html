function badge(text, bg, color) { return `<span class="badge" style="background:${bg};color:${color}">${text}</span>`; }

function statusBadge(text, isUrgent) {

  const bg = isUrgent ? '#fff0ea' : '#f1f1ed';

  const color = isUrgent ? '#a53c1e' : '#575750';

  return `<span class="status" style="background:${bg};color:${color}">${text}</span>`;

}

function deadlineItem(d) {

  const p = P[d.prj];

  const cls = d.urgent ? 'urgent' : '';

  return `<div class="item ${cls}">

    <div class="date" style="color:${p.text}">${d.date}</div>

    <div>

      <div class="name">${d.name}${badge(p.label, p.bg, p.text)}</div>

      <div class="meta">담당: ${d.assignee} · ${d.note}</div>

    </div>

    ${statusBadge(d.status, d.urgent)}

  </div>`;

}

function personCard(p) {

  return `<div class="person">

    <div class="person-head">

      <div class="avatar" style="background:${p.bg};color:${p.color}">${p.name[0]}</div>

      <div><div class="person-name">${p.name}</div><div class="person-role">${p.role}</div></div>

    </div>

  </div>`;

}

function renderProject(key) {

  const list = key === 'all' ? deadlines : deadlines.filter(d => d.prj === key);

    const active = btn.dataset.project === key;

    btn.style.color = active ? p.text : '#111';

  });

}
